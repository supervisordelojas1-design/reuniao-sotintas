/* ============================================================================
   ACESSO · PIN ÚNICO — Assistente ST · Só Tintas · novembro/2026

   O PROBLEMA QUE ISTO RESOLVE
   O app tem dois servidores. O do app (agenda, aba USUARIOS) é o único que
   grava PIN: login, "trocar PIN" e "liberar / novo PIN" escrevem lá. O
   Servidor de Dados é quem emite o CRACHÁ que abre PDI, Níveis, Ranking,
   Mentor, Painel de Metas e CRM — e ele confere o PIN na base dele.

   Enquanto as duas bases não forem a mesma, acontece isto:
     dia 0  colaborador entra com 1234 (as duas bases têm 1234) -> crachá, 7 dias
     dia 0  ele troca o PIN -> a troca vai SÓ para a base do app
     1 a 6  tudo funciona: o crachá antigo ainda está válido
     dia 7  o crachá vence, o app pede o PIN, ele digita o PIN que cadastrou
            e ESTE servidor recusa, porque aqui ainda está o 1234. Travado.

   COMO USAR (3 minutos)
   1. Abra o Apps Script do SERVIDOR DE DADOS (o que emite o crachá).
   2. Arquivos -> + -> Script -> nome "acesso-pin-unico" -> cole tudo isto.
   3. Preencha PIN_FONTE abaixo com o ID (ou a URL) da planilha do APP.
   4. Escolha a função  stPinDiagnostico  -> Executar. Leia o registro: ele
      diz se as bases são a mesma e lista nome por nome quem está divergente.
   5. Escolha a função  stPinSincronizar  -> Executar. Destrava todo mundo.
   6. Escolha a função  stPinGatilhoDiario -> Executar (uma vez). A partir daí
      qualquer troca de PIN no app chega aqui sozinha, de madrugada.
   7. Para o reparo valer na hora (sem esperar a madrugada), ligue a ação
      sincPin: no Código.gs, dentro do doPost, logo antes do "switch (d.acao) {",
      acrescente UMA linha:

          if (d.acao === 'sincPin') return stSincPinAcao_(d);

      Se o doPost devolver com ContentService, use o mesmo padrão das outras
      ações do arquivo (veja a nota em stSincPinAcao_).

   SEGURO PARA RODAR DUAS VEZES: na segunda ele encontra tudo igual e não
   escreve nada. Não cria, não apaga e não desativa ninguém — só acerta o PIN
   de quem já existe nas duas bases, e lista quem falta para você decidir.
   ============================================================================ */

/* ID (ou URL completa) da planilha do APP — a que tem a aba USUARIOS onde o
   login e a troca de PIN gravam. Deixe '' para usar a planilha deste script. */
var PIN_FONTE = '';

var PIN_ABA = 'USUARIOS';   /* nome da aba nas duas planilhas */


/* ---------- 1) DIAGNÓSTICO: prova onde está o problema, sem mudar nada ---------- */
function stPinDiagnostico() {
  var log = [], diz = function (t) { log.push(t); Logger.log(t); };
  var f, d;
  try { f = stPinBase_(stPinFonte_()); } catch (e) { diz('PAREI: fonte — ' + e.message); return log.join('\n'); }
  try { d = stPinBase_(SpreadsheetApp.getActive()); } catch (e) { diz('PAREI: base local — ' + e.message); return log.join('\n'); }

  diz('FONTE (app) ....... ' + f.ss.getName() + '  [' + f.ss.getId() + ']  ' + f.linhas.length + ' pessoas');
  diz('ESTE servidor ..... ' + d.ss.getName() + '  [' + d.ss.getId() + ']  ' + d.linhas.length + ' pessoas');

  if (f.ss.getId() === d.ss.getId()) {
    diz('');
    diz('As duas bases são a MESMA planilha: nada para sincronizar.');
    diz('Então a recusa do PIN no 7º dia vem de outro ponto. Confira, no Código.gs:');
    diz('  · a ação "entrar" está lendo a coluna PIN desta mesma aba USUARIOS?');
    diz('  · ela compara o nome normalizado (sem acento, minúsculo, espaços)?');
    diz('  · o código AUTCOM também é aceito no lugar do nome?');
    return log.join('\n');
  }

  var iguais = 0, divergentes = [], soNaFonte = [], soAqui = [];
  f.linhas.forEach(function (p) {
    var q = d.mapa[p.chave];
    if (!q) { soNaFonte.push(p.nome); return; }
    if (String(q.pin) === String(p.pin)) iguais++;
    else divergentes.push(p.nome + '  (app: ' + stPinMasc_(p.pin) + '  ·  aqui: ' + stPinMasc_(q.pin) + ')');
  });
  d.linhas.forEach(function (q) { if (!f.mapa[q.chave]) soAqui.push(q.nome); });

  diz('');
  diz('PIN igual nas duas ................. ' + iguais);
  diz('PIN DIVERGENTE (estes travam) ..... ' + divergentes.length);
  divergentes.forEach(function (t) { diz('   · ' + t); });
  diz('Está no app e não está aqui ....... ' + soNaFonte.length);
  soNaFonte.forEach(function (t) { diz('   · ' + t); });
  diz('Está aqui e não está no app ....... ' + soAqui.length);
  soAqui.forEach(function (t) { diz('   · ' + t); });
  diz('');
  diz(divergentes.length ? 'Rode stPinSincronizar para acertar os divergentes.' : 'Nada divergente: ninguém travado por PIN agora.');
  return log.join('\n');
}


/* ---------- 2) SINCRONIZAR: a fonte manda, este servidor obedece ---------- */
function stPinSincronizar() {
  var log = [], diz = function (t) { log.push(t); Logger.log(t); };
  var f = stPinBase_(stPinFonte_()), d = stPinBase_(SpreadsheetApp.getActive());

  if (f.ss.getId() === d.ss.getId()) { diz('Mesma planilha: nada a fazer.'); return log.join('\n'); }

  var mudou = 0, faltam = [];
  f.linhas.forEach(function (p) {
    var q = d.mapa[p.chave];
    if (!q) { faltam.push(p.nome); return; }
    if (String(q.pin) === String(p.pin)) return;
    d.aba.getRange(q.linha, d.cPin + 1).setValue(String(p.pin));
    diz('  ' + p.nome + ': ' + stPinMasc_(q.pin) + ' -> ' + stPinMasc_(p.pin));
    mudou++;
  });
  SpreadsheetApp.flush();
  diz('');
  diz('PINs acertados: ' + mudou);
  if (faltam.length) {
    diz('Sem cadastro NESTE servidor (ninguém mexeu neles, confira se devem ter acesso aos painéis):');
    faltam.forEach(function (t) { diz('   · ' + t); });
  }
  diz(mudou ? 'Quem estava travado já entra com o PIN que cadastrou.' : 'Tudo já estava igual.');
  return log.join('\n');
}


/* ---------- 3) GATILHO DIÁRIO ---------- */
function stPinGatilhoDiario() {
  stPinRemoverGatilho();
  ScriptApp.newTrigger('stPinSincronizar').timeBased().atHour(3).everyDays(1).create();
  Logger.log('Gatilho criado: stPinSincronizar todo dia por volta das 3h.');
  return 'ok';
}
function stPinRemoverGatilho() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'stPinSincronizar') ScriptApp.deleteTrigger(t);
  });
  return 'ok';
}


/* ---------- 4) AÇÃO sincPin: reparo na hora, chamada pelo app ----------
   O app chama isto quando este servidor recusa um PIN que a aba USUARIOS
   aceita. Nós NÃO confiamos no que vem na chamada: conferimos o PIN na FONTE
   antes de gravar. Quem não confere na fonte não muda nada aqui.

   NOTA: se o seu doPost devolve as respostas com
         ContentService.createTextOutput(JSON.stringify(x)).setMimeType(...),
   troque a linha do passo 7 por:
         if (d.acao === 'sincPin') return stPinSaida_(stSincPinAcao_(d));
   ------------------------------------------------------------------------- */
function stSincPinAcao_(d) {
  try {
    var nome = String((d && d.nome) || '').trim(), pin = String((d && d.pin) || '').trim();
    if (!nome || !/^\d{4,6}$/.test(pin)) return { ok: false, erro: 'Dados incompletos.' };

    var f = stPinBase_(stPinFonte_()), chave = stPinChave_(nome);
    var naFonte = f.mapa[chave];
    if (!naFonte) return { ok: false, erro: 'Não achei esse nome no cadastro do app.' };
    if (String(naFonte.pin) !== pin) return { ok: false, erro: 'O PIN não confere no cadastro do app.' };

    var dd = stPinBase_(SpreadsheetApp.getActive()), aqui = dd.mapa[chave];
    if (!aqui) return { ok: false, erro: 'Esse nome não tem cadastro no servidor dos painéis.' };
    if (String(aqui.pin) !== pin) {
      dd.aba.getRange(aqui.linha, dd.cPin + 1).setValue(pin);
      SpreadsheetApp.flush();
    }
    return { ok: true, nome: naFonte.nome };
  } catch (e) {
    return { ok: false, erro: String(e && e.message || e) };
  }
}
function stPinSaida_(x) {
  return ContentService.createTextOutput(JSON.stringify(x)).setMimeType(ContentService.MimeType.JSON);
}

/* ---------- peças internas ---------- */
function stPinFonte_() {
  var v = String(PIN_FONTE || '').trim();
  if (!v) return SpreadsheetApp.getActive();
  var m = /[-\w]{25,}/.exec(v);              /* aceita ID ou URL completa */
  if (!m) throw new Error('PIN_FONTE não parece um ID nem uma URL de planilha.');
  return SpreadsheetApp.openById(m[0]);
}
function stPinChave_(s) {
  return String(s || '').normalize('NFD').replace(/[̀-ͯ]/g, '')
    .toLowerCase().replace(/\s+/g, ' ').trim();
}
function stPinMasc_(p) {
  p = String(p == null ? '' : p);
  return p ? p.charAt(0) + new Array(p.length).join('•') : '(vazio)';
}
/* lê a aba USUARIOS pelos cabeçalhos, não por posição de coluna */
function stPinBase_(ss) {
  var aba = ss.getSheetByName(PIN_ABA);
  if (!aba) throw new Error('não achei a aba ' + PIN_ABA + ' em "' + ss.getName() + '".');
  var vals = aba.getDataRange().getValues();
  if (!vals.length) throw new Error('a aba ' + PIN_ABA + ' de "' + ss.getName() + '" está vazia.');

  var cab = vals[0].map(function (c) { return stPinChave_(c); });
  var acha = function (nomes) {
    for (var i = 0; i < cab.length; i++) if (nomes.indexOf(cab[i]) >= 0) return i;
    return -1;
  };
  var cNome = acha(['nome', 'colaborador', 'usuario', 'usuário']);
  var cPin = acha(['pin', 'senha']);
  if (cNome < 0) throw new Error('não achei a coluna NOME na aba ' + PIN_ABA + ' de "' + ss.getName() + '".');
  if (cPin < 0) throw new Error('não achei a coluna PIN na aba ' + PIN_ABA + ' de "' + ss.getName() + '".');

  var linhas = [], mapa = {};
  for (var r = 1; r < vals.length; r++) {
    var nome = String(vals[r][cNome] || '').trim();
    if (!nome) continue;
    var reg = { nome: nome, chave: stPinChave_(nome), pin: String(vals[r][cPin] == null ? '' : vals[r][cPin]).trim(), linha: r + 1 };
    linhas.push(reg);
    if (!mapa[reg.chave]) mapa[reg.chave] = reg;   /* o primeiro manda: duplicado não sobrescreve */
  }
  return { ss: ss, aba: aba, cNome: cNome, cPin: cPin, linhas: linhas, mapa: mapa };
}
