var TOKEN = 'sotintas-pdi-2026';     // o mesmo de pdi.html
var PEDACO = 45000;                  // célula do Sheets aceita até 50 mil caracteres

function aba_(nome, cab) {
  var ss = SpreadsheetApp.getActive();
  var sh = ss.getSheetByName(nome);
  if (!sh) { sh = ss.insertSheet(nome); sh.appendRow(cab); sh.setFrozenRows(1); }
  return sh;
}
function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

/* ---------- leitura ---------- */
function lerBase_() {
  var sh = aba_('BASE', ['parte', 'conteudo', 'publicado_em', 'por']);
  var n = sh.getLastRow(); if (n < 2) return { base: null, atualizado: null };
  var v = sh.getRange(2, 1, n - 1, 4).getValues().sort(function (a, b) { return a[0] - b[0]; });
  var txt = v.map(function (r) { return String(r[1]); }).join('');
  return { base: txt ? JSON.parse(txt) : null, atualizado: v[0][2] ? new Date(v[0][2]).toISOString() : null, por: v[0][3] || '' };
}
function lerAcomp_() {
  var sh = aba_('ACOMP', ['chave', 'loja', 'estado', 'por', 'atualizado_em']);
  var n = sh.getLastRow(), out = {}; if (n < 2) return out;
  sh.getRange(2, 1, n - 1, 5).getValues().forEach(function (r) { if (r[0]) { try { out[r[0]] = JSON.parse(r[2]); } catch (e) {} } });
  return out;
}

function doGet(e) {
  var p = (e && e.parameter) || {};
  try {
    if (p.token !== TOKEN) return json_({ ok: false, erro: 'acesso negado' });
    if (p.acao === 'base') { var b = lerBase_(); return json_({ ok: true, base: b.base, atualizado: b.atualizado, por: b.por }); }
    if (p.acao === 'acomp') return json_({ ok: true, acomp: lerAcomp_() });
    if (p.acao === 'diag') { var b2 = lerBase_(); return json_({ ok: true, vendedores: b2.base ? b2.base.V.length : 0, atualizado: b2.atualizado, acompanhamentos: Object.keys(lerAcomp_()).length }); }
    return json_({ ok: false, erro: 'ação desconhecida' });
  } catch (err) { return json_({ ok: false, erro: String(err) }); }
}

/* ---------- gravação ---------- */
function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(20000);
    var d = JSON.parse(e.postData.contents || '{}');
    if (d.token !== TOKEN) return json_({ ok: false, erro: 'acesso negado' });

    if (d.acao === 'publicar') {
      if (!d.base || !d.base.V || !d.base.V.length) return json_({ ok: false, erro: 'base vazia' });
      var sh = aba_('BASE', ['parte', 'conteudo', 'publicado_em', 'por']);
      if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 4).clearContent();
      var txt = JSON.stringify(d.base), agora = new Date(), linhas = [];
      for (var i = 0, k = 1; i < txt.length; i += PEDACO, k++) linhas.push([k, txt.substr(i, PEDACO), agora, d.por || '']);
      sh.getRange(2, 1, linhas.length, 4).setValues(linhas);
      return json_({ ok: true, partes: linhas.length, vendedores: d.base.V.length });
    }

    if (d.acao === 'salvar') {
      if (!d.chave || !d.loja) return json_({ ok: false, erro: 'chave ou loja ausente' });
      var est = JSON.stringify(d.estado || {});
      if (est.length > 45000) return json_({ ok: false, erro: 'registro grande demais' });
      var sa = aba_('ACOMP', ['chave', 'loja', 'estado', 'por', 'atualizado_em']);
      var n = sa.getLastRow(), linha = 0;
      if (n > 1) { var ch = sa.getRange(2, 1, n - 1, 1).getValues(); for (var j = 0; j < ch.length; j++) if (ch[j][0] === d.chave) { linha = j + 2; break; } }
      var reg = [d.chave, "'" + d.loja, est, d.por || '', new Date()];
      if (linha) sa.getRange(linha, 1, 1, 5).setValues([reg]); else sa.appendRow(reg);
      return json_({ ok: true });
    }
    return json_({ ok: false, erro: 'ação desconhecida' });
  } catch (err) {
    return json_({ ok: false, erro: String(err) });
  } finally { try { lock.releaseLock(); } catch (e2) {} }
}
